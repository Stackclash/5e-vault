> >     }
> >     levelText += `>> ${classSpellConfig.spells.filter(spell => spell.level === (level)).map(spell => spell.name).join(`\n>>\n>> `)}`
> >     result.push(levelText)
> >   })
> > })
> > dv.paragraph(`> [!cards|wfull]\n${result.join(`\n>\n`)}`)
